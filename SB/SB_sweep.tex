\begin{frame}{Recap — Fokker--Planck Formulation}

Let $\mu,\nu\in\mathcal{P}(\mathbb{R}^d)$ be the initial and terminal marginals,
time rescaled to $[0,1]$, $\gamma>0$ the diffusion coefficient.

We seek a density--velocity pair $(\rho_t,\bt_t)_{t\in[0,1]}$ with
$\rho_0=\mu$, $\rho_1=\nu$ such that
\begin{itemize}
    \item $\rho_t$ solves the \textbf{Fokker--Planck equation}
    \[
        \partial_t \rho_t + \dive(\rho_t\,\bt_t) = \gamma\,\Delta \rho_t ,
    \]
    \item the pair minimizes the \textbf{expected kinetic energy}
    \[
        \mathcal{J}(\rho,\bt)=\int_0^1\!\!\int_{\mathbb{R}^d}\tfrac12|\bt_t(\xv)|^2\,\rho_t(\xv)\,d\xv\,dt .
    \]
\end{itemize}

Girsanov applied to $dX_t=\bt_t(X_t)\,dt+\sqrt{2\gamma}\,dW_t$ gives
$\mathrm{KL}(\mathbb{P}\,\|\,\mathbb{R})=\mathcal{J}/(2\gamma)$: the Schrödinger
bridge is the minimal-energy stochastic interpolation between $\mu$ and $\nu$.

\end{frame}
\begin{frame}{Recap — Heat and Advection--Diffusion Semigroups}

Hopf--Cole ($\eta=e^{f}$, $\eta^\ast=e^{g}$) linearizes the optimality system into
two transport--diffusion PDEs, whose transition operators are the SB kernel.

\begin{block}{Driftless bridge ($\boldsymbol{\beta}\equiv0$) — heat semigroup $\Pcal_\tau=e^{\tau\gamma\Delta}$}
\[
    \partial_\tau\phi=\gamma\,\Delta\phi,\qquad \partial_n\phi|_{\partial\Omega}=0
    \quad(\text{Neumann, no-flux}).
\]
\end{block}

\begin{block}{Drifted bridge — advection--diffusion semigroups $\Qcal_\tau=e^{\tau\mathcal{L}}$, $\Qcal^{\dagger}_\tau=e^{\tau\mathcal{L}^\ast}$}
\[
    \underbrace{\partial_\tau\phi=\boldsymbol{\beta}\cdot\grad\phi+\gamma\Delta\phi}_{\Qcal:\ \text{non-conservative, backward}},
    \qquad
    \underbrace{\partial_\tau\phi=-\dive(\boldsymbol{\beta}\phi)+\gamma\Delta\phi}_{\Qcal^{\dagger}:\ \text{conservative, forward}} .
\]
\end{block}

IPFP in log-potential form:
\[
    f^{(k+1)}=\log\mu-\log\Qcal_1\!\bigl[e^{g^{(k)}}\bigr],
    \qquad
    g^{(k+1)}=\log\nu-\log\Qcal^{\dagger}_1\!\bigl[e^{f^{(k+1)}}\bigr].
\]
FVM: TPFA diffusion $+$ first-order upwind advection, SSP-RK2, Anderson-accelerated.
$\boldsymbol{\beta}\equiv0\Rightarrow\Qcal_1=\Qcal^{\dagger}_1=\Pcal_1$.

\end{frame}
\begin{frame}{Recap — Convex Displacement Interpolation (CDI)}

The maps act on the shock-feature marginals; the physical Mach field is recovered
gradient-free. For $t\in(0,1)$,
\[
    \boxed{\;
    \widehat M(t,\xv)=(1-t)\,M_0\!\bigl((1-t)\,\xv+t\,S(\xv)\bigr)
                     +t\,M_1\!\bigl(t\,\xv+(1-t)\,T(\xv)\bigr)
    \;}
\]

\begin{itemize}
    \item $M_0,M_1$ = endpoint Mach fields, sampled at the displaced pre-images by
    linear interpolation on the mesh;
    \item exact at the endpoints: $\widehat M(0,\cdot)=M_0$, $\widehat M(1,\cdot)=M_1$;
    \item no map inversion — forward pre-image $t\xv+(1-t)T$ pulls back $M_1$,
    backward pre-image $(1-t)\xv+tS$ pulls back $M_0$, blended with weights $(1-t,t)$.
\end{itemize}

The reference drift is designed to keep $T,S$ close to the identity, so the CDI
reconstruction stays well-conditioned at small $\gamma$.

\end{frame}
\begin{frame}{Recap — Construction of the Transport Maps $T,S$}

$T$ is the conditional mean of the target position given the source (built from
$g$ through $\Qcal_1$); $S$ is its backward analogue (from $f$ through $\Qcal^{\dagger}_1$):
\[
    T(\xv)=\frac{\Qcal_1[\,\id\,e^{g}\,](\xv)}{\Qcal_1[e^{g}](\xv)}-\boldsymbol{\beta}_{\mathrm{bias}}(\xv),
    \qquad
    S(\xv)=\frac{\Qcal^{\dagger}_1[\,\id\,e^{f}\,](\xv)}{\Qcal^{\dagger}_1[e^{f}](\xv)}-\boldsymbol{\beta}_{\mathrm{bias}}(\xv).
\]

\begin{block}{Neumann identity-leak correction}
\[
    \boldsymbol{\beta}_{\mathrm{bias}}=\frac{\Qcal_1[\id]}{\Qcal_1[1]}-\Phi_{\bt},
\]
measured against the drift's own \emph{reflected deterministic flow} $\Phi_{\bt}$
(identity when $\bt\equiv0$), so $\bt$'s transport stays in $T,S$.
\end{block}

Evaluated in the stable log-domain with a positivity shift $\id\mapsto\id-c$; a
trust gate reverts kernel-collapse outlier cells to the identity. $T,S\to\id$ far
from the transported feature.

\end{frame}
\begin{frame}{Oblique Drift (1/2) — Wave Angles}

The LE and TE shocks rotate between their $M_0$ and $M_1$ wave angles:

\begin{block}{$\theta$--$\beta$--$M$ relation (weak root) — LE family}
\[
    \tan\theta = 2\cot\beta_s\,\frac{M^2\sin^2\beta_s - 1}{M^2(\gamma_g+\cos2\beta_s)+2},
    \qquad
    \beta_s^{\mathrm{LE}}(M) = \beta_s(\alpha, M).
\]
\end{block}

\begin{block}{TE family — LE shock $\to$ Prandtl--Meyer($2\alpha$) $\to$ TE shock}
\[
    \nu(M_b) = \nu(M_a) + 2\alpha,
    \qquad
    \beta_s^{\mathrm{TE}}(M) = \beta_s(\alpha, M_b(M)).
\]
\end{block}

\begin{block}{Ray angles and rotation rates,\; $M(t)=(1-t)M_0+tM_1$}
\[
    \psi_{\mathrm{LE}} = \beta_s^{\mathrm{LE}}, \qquad
    \psi_{\mathrm{TE}} = \beta_s^{\mathrm{TE}}-\alpha, \qquad
    \omega_\kappa(t) = \tfrac{d\beta_s^{\kappa}}{dM}\big|_{M(t)}\,(M_1-M_0).
\]
\end{block}

\end{frame}
\begin{frame}{Oblique Drift (2/2) — Windowed Rigid Rotation}

\begin{block}{Four windowed rigid rotations about the apexes $\av_\kappa$}
\[
    \boxed{\;
    \boldsymbol{\beta}(t,\xv) = \sum_{\kappa\in\{\mathrm{LE},\mathrm{TE}\}}\sum_{\sigma=\pm1}
        \sigma\,\omega_\kappa(t)\,w_{\kappa,\sigma}(t,\xv)\,\R\,(\xv-\av_\kappa),
    \qquad
    \R = \bigl(\begin{smallmatrix}0&-1\\1&0\end{smallmatrix}\bigr)
    \;}
\]
\[
    w_{\kappa,\sigma} = e^{-\mathrm{dist}_{\kappa,\sigma}^2/\sigma_w(s)^2}\,
        \mathbf{1}[s_{\kappa,\sigma}>0],
    \qquad
    s_{\kappa,\sigma} = (\xv-\av_\kappa)\cdot\dhat_{\kappa,\sigma}.
\]
\end{block}

Zero in the freestream; the conical band $\sigma_w(s)$ covers the swept wedge;
$\dive\boldsymbol{\beta}\neq0$ $\Rightarrow$ enters through the conservative $\Qcal^{\dagger}$.

\end{frame}

\begin{frame}
  \frametitle{Free-Form Deformation (FFD)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{align*}
        T: \mathbb{R}^{N_{cp}} \times \mathbb{R}^3 &\longrightarrow
        \mathbb{R}^3 \\
        \lambda, \xi &\longmapsto T_{\lambda_0}(\xi) = x
      \end{align*}
      In the coordinate system $\mathbf{e_1}$, $\mathbf{e_2}$, $\mathbf{e_3}$

      $\mathbf{x} = \mathbf{x_0} + t_1 \mathbf{e_1} + t_2
      \mathbf{e_2} + t_3 \mathbf{e_3}$

      The mapping is defined as:
      \begin{equation}
        FFD(t_1,t_2,t_3) = \sum_{i=0}^{l} \sum_{j=0}^{m} \sum_{k=0}^{n}
        B_{i}^{l}(t_1) B_{j}^{m}(t_2) B_{k}^{n}(t_3) P_{i,j,k}
      \end{equation}

      Here, $P_{i,j,k}$ are the control points in the deformed lattice,
      and $l, m, n$ denote the degree of the polynomial in each direction.

    \end{column}
    % \begin{column}{0.3\textwidth}
    %   In the coordinates system $\mathbf{e_1}$, $\mathbf{e_2}$, $\mathbf{e_3}$

    %   $\mathbf{x} = \mathbf{x_0} + t_1 \mathbf{e_1} + t_2
    % \mathbf{e_2} + t_3 \mathbf{e_3}$
    % \end{column}
    \begin{column}{0.5\textwidth}
      {\small
        The $(t_1, t_2, t_3)$ coordinates can be calculated using:

        \begin{align}
          t_1 &= \frac{(\mathbf{e_2} \times \mathbf{e_3}) \cdot (X -
          X_0)}{(\mathbf{e_2} \times \mathbf{e_3}) \cdot \mathbf{e_1}} \\
          t_2 &= \frac{(\mathbf{e_1} \times \mathbf{e_3}) \cdot (X -
          X_0)}{(\mathbf{e_1} \times \mathbf{e_3}) \cdot \mathbf{e_2}} \\
          t_3 &= \frac{(\mathbf{e_1} \times \mathbf{e_2}) \cdot (X -
          X_0)}{(\mathbf{e_1} \times \mathbf{e_2}) \cdot \mathbf{e_3}}
        \end{align}

        Where a Bernstein polynomial is defined as:

        \begin{equation}
          B_{i}^{l}(t) = \binom{l}{i}t^{i}(1-t)^{l-i}
        \end{equation}
      }
    \end{column}
  \end{columns}
\end{frame}
\begin{frame}{Data-Driven Drift — FFD Registration}

Register the two shock-feature marginals $u_0,u_1$ by two FFD maps $T$ (forward)
and $W$ (backward), whose control-point displacements $\Delta P^{T},\Delta P^{W}$
are the free DOF (single region, all control points free, no boundary conditions).

\begin{block}{Objective (single region), weights $(\alpha_A,\alpha_B,\alpha_C,\alpha_D,\alpha_G)=(1,\,0.1,\,1,\,10^{-2},\,0.1)$}
{\small
\begin{align*}
J=\;&\alpha_A\bigl(\|u_1\!\circ T-u_0\|^2+\|u_0\!\circ W-u_1\|^2\bigr)\\
 +\;&\alpha_B\bigl(\|W\!\circ T-\id\|^2+\|T\!\circ W-\id\|^2\bigr)\\
 +\;&\alpha_C\bigl(\|b_\varepsilon(\det\grad T)\|^2+\|b_\varepsilon(\det\grad W)\|^2\bigr)
    +\tfrac{\alpha_D}{N_{cp}}\bigl(\|\Delta P^{T}\|^2+\|\Delta P^{W}\|^2\bigr)\\
 +\;&\alpha_G\bigl(\mathcal{W}_2^2(u_1\!\circ T,u_0)+\mathcal{W}_2^2(u_0\!\circ W,u_1)\bigr)
\end{align*}}
\end{block}

{\footnotesize
$\alpha_A$ alignment (both directions)\; · \;$\alpha_B$ bijectivity / map-sync \;
· \;$\alpha_C$ Jacobian log-barrier (positivity) \; · \;$\alpha_D$ control-point reg.\ \;
· \;$\alpha_G$ $\mathcal{W}_2$ capture.\par
$T,W$ analytic (Bernstein) $\Rightarrow$ evaluated \emph{exactly} at the true cell
centres — no grid$\to$mesh interpolation error.}

\end{frame}
\begin{frame}{From FFD Mappings to the Reference Drift $\boldsymbol{\beta}$}

The alignment term drives $u_1(T(\xv))=u_0(\xv)$: at a field-0 feature $p_0$,
$T(p_0)=p_1$ (its field-1 position). Evaluating the converged Bernstein maps at the
true cell centres and taking the frozen unit-scale ($s{=}1$) displacements,

\[
    \boxed{\;\boldsymbol{\beta}(\xv)=T(\xv,s{=}1)-\xv\;}
    \qquad
    \boldsymbol{\alpha}(\xv)=W(\xv,s{=}1)-\xv
\]
\[
    \text{forward } (\mu_0\!\to\!\mu_1)\ \text{via }\boldsymbol{\beta};
    \qquad
    \text{backward } (W\approx T^{-1})\ \text{via }\boldsymbol{\alpha}.
\]

\begin{itemize}
    \item $\boldsymbol{\beta}$ is a frozen $(N,2)$ field, consumed by the SB kernel
    $\Qcal/\Qcal^{\dagger}$, $\Phi_{\bt}$-referenced bias, trust gate and CDI
    \emph{exactly} like the oblique drift;
    \item $\boldsymbol{\beta}$ is cached per marginal pair
    (\texttt{ffd\_beta\_$\langle$hash$\rangle$.npz}) — the registration is offline,
    run once;
    \item the raw maps $T=\xv+\boldsymbol{\beta}$, $S=\xv+\boldsymbol{\alpha}$ also
    serve as a \emph{stand-alone} interpolator (FFD control).
\end{itemize}

\end{frame}

\section{Self-Conditioned Drift}


\begin{frame}{$\mathrm{SB}^2$ — Self-Conditioned Drift}

Take the reference drift from the \emph{previous} $\gamma$-stage's own solution,
carrying its full time dependence:
\[
  \gamma_0>\gamma_1>\dots>\gamma_K,
  \qquad
  \boldsymbol{\beta}^{(k+1)}(t)\;=\;\boldsymbol{\beta}^{(k)}(t)\;+\;2\gamma_k\,\grad g^{(k)}_t ,
\]
stored as a $K\times N\times 2$ stack and fed to the kernel at every substep.

\vspace{0.4em}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Why it is well posed}
$\boldsymbol{\beta}$ optimal $\Rightarrow\grad g\to0\Rightarrow
\boldsymbol{\beta}'=\boldsymbol{\beta}$: the SB solution is a fixed point of the map.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Time direction}
$\Qcal$ reads $\boldsymbol{\beta}$ \emph{backward} ($1-\tau$),
$\Qcal^{\dagger}$ \emph{forward} ($\tau$).\\
Mirrored: $\langle\Qcal u,v\rangle=\langle u,\Qcal^{\dagger}v\rangle$ to
$7\times10^{-15}$.
\end{alertblock}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{$\mathrm{SB}^2$ — Conditioning: FFD Seed and Endpoint Clip}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{1. Seed and switch}
$\boldsymbol{\beta}^{(0)}=\boldsymbol{\beta}_{\mathrm{FFD}}$, held \textbf{fixed}
while $\gamma>\gamma_\star=10^{-3}$; self-conditioning below.\\[3pt]
{\small A heat-seeded start recovers a large, void-dominated $2\gamma\grad g$ and
loses the whole mid-ladder.}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{2. Endpoint clip}
$g_t=\log\Qcal_{1-t}[e^{g}]$ and $\Qcal_0=\mathrm{Id}$, so at $t=1$ the
\emph{raw} log-potential's gradient enters $\boldsymbol{\beta}$.\\[3pt]
{\small Sample $t\in[c,1-c]$ with $c=0.2$.}
\end{block}
\end{column}
\end{columns}

\begin{frame}{Summary}
\begin{itemize}
  \item \textbf{Regime dominates.} Error falls $\sim\!2$ decades from
        $M\!\approx\!0.85$ to $2.95$; sub-$M1$ bands are $30\!-\!100\times$ worse
        and lie outside the attached-shock premise the marginal is built on.
  \item \textbf{Linear wins above $M\approx2.45$} --- transport buys nothing once
        the shock barely moves across the band.
  \item \textbf{On the AoA axis SB loses to Linear by $4\!-\!8\times$}, and the
        FFD registration recovers $\approx\!0$ displacement (FFD $\equiv$ Linear).
  \item \textbf{\texttt{oblique} is the best drift} at every $\gamma$
        ($+3\!-\!7\%$, $p<10^{-4}$); \texttt{null} is $+26\%$ worse and unstable
        at high Mach.
  \item \textbf{The raw FFD registration still beats the bridge built on it}
        ($1.24\!\times\!10^{-3}$ vs $1.72\!\times\!10^{-3}$ on matched bands).
\end{itemize}\end{frame}
